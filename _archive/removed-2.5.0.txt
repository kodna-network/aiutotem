AIU TOTEM 2.5.0 - what left the app (nothing is deleted for good; this is how to bring it back)
=============================================================================================

1. THE JOY STREAK (js/life.js, 2.4.0). Replaced by tokens and the Corner Mart. The x2 rule in core.js still counts
   for events made between 2.4.0 (C.T240) and 2.5.0 (C.T250), so past growth never changes. L.joy, L.x2, L.mark,
   L.joyFill and L.joyLeft are kept as no-ops so older calls keep working. The original block:

  /* ---------- the joy streak ---------- */
  var JOY_FULL = 20, JOY_MS = 3 * 60000, GAIN = { pet: 1, snack: 3, water: 3, help: 4, step: 3, game: 4, tile: 5 };
  var joy = { pts: 0, x2Until: 0, petAt: 0 };
  L.x2 = function () { return Date.now() < joy.x2Until; };
  L.joyLeft = function () { return Math.max(0, joy.x2Until - Date.now()); };
  L.joyFill = function () { return L.x2() ? 1 : Math.min(1, joy.pts / JOY_FULL); };
  L.joy = function (kind) {
    var a = app(); if (!a.S || !a.S.hatched) return;
    var now = Date.now();
    if (kind === 'pet') { if (now - joy.petAt < 3000) return; joy.petAt = now; }
    if (L.x2()) return draw();
    joy.pts += GAIN[kind] || 1;
    if (joy.pts >= JOY_FULL) {
      joy.pts = 0; joy.x2Until = now + JOY_MS;
      try { a.beep('ping'); ui().hearts(5, ['star', 'heart', 'sparkle']); ui().k.snack('Joy streak! Everything counts double for 3 minutes.'); } catch (e) {}
    }
    draw();
  };
  // the little star meter beside the heart and the leaf
  var chip = null;
  L.chip = function () {
    chip = h('button', { type: 'button', cls: 'joychip', 'aria-label': 'Joy streak', onclick: function () { ui().k.sheet('Joy streak', [
      h('p', { text: 'Pet ' + app().S.totemName + ', feed and water the pet, drag him to what he needs, check off steps and win games. When the star fills up, everything that grows him counts double for 3 minutes.' }),
      h('p', { cls: 'small', text: 'Petting counts once every few seconds. The daily growth limit still holds: the streak just gets you there faster.' })]); } },
      h('span', { cls: 'jstar' }, ui().pix('star')), h('span', { cls: 'jbar' }, h('i')), h('b', { cls: 'jx' }));
    draw(); return chip;
  };
  function draw() {
    if (!chip) return;
    var on = L.x2(), left = Math.ceil(L.joyLeft() / 1000);
    chip.classList.toggle('on', on);
    chip.querySelector('.jbar i').style.width = Math.round(L.joyFill() * 100) + '%';
    chip.querySelector('.jx').textContent = on ? 'x2 ' + Math.floor(left / 60) + ':' + U.p2(left % 60) : '';
    chip.setAttribute('aria-label', on ? 'Joy streak: double growth for ' + left + ' more seconds' : 'Joy streak ' + Math.round(L.joyFill() * 100) + '% full');
  }
  setInterval(function () { if (chip && L.x2()) draw(); else if (chip && chip.classList.contains('on')) draw(); }, 1000);
  // events that can count double (marked when they're made, so every device agrees)
  var X2 = { pet: 1, quest_step: 1, tile_done: 1, game: 1, gaba: 1, journal: 1, jpage: 1 };
  L.mark = function (d) {
    if (!d || !X2[d.type]) return d;
    if (d.type === 'quest_step' && d.data && d.data.done) L.joy('step');
    if (d.type === 'tile_done' && d.data && !d.data.undone) L.joy('tile');
    if (L.x2()) d.data = Object.assign({}, d.data || {}, { x2: 1 });
    return d;
  };

3. ACCOUNTS > THE PROFILE PAGE'S THREE BIG BUTTONS (js/business.js B.profileView, 2.3.1): My info (name, birthday,
   address, About me), Docs, Money. The code is still in business.js below the 2.5.0 early return
   ("if (AIU.panels.accounts) return AIU.panels.accounts();"): delete that line to bring the old page back. Accounts
   is four tabs now: Cards, Person (the personality parts of About me), Files (your files + the Written folder with
   the old Docs list) and Finance (the old Money).
   Your private details card ("Me", v-me) is no longer made automatically (accounts.js ensureMe is kept, unused).
   An existing Me card stays, as an ordinary card. The profile card at the top of Accounts (profiles.js PF.card) is
   kept, unused.

   your username. The setting (callMe) is kept and still used by device-only Aius.
