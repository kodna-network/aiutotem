=== api.php: admin recovery ===
// ------------------------------------------------------------------ 2.6.0: admin recovery. Someone locked out contacts the admin;
// the admin sets a temporary password here, tells them, and they change it in Settings. Every device is logged out.
// Their account cards and files open again with their own recovery code (made on their device; the server never has it).
function act_admin_recover($req) {
  rate_limit('admin', 8, 3600);
  $ok = hash_equals(strtolower(ADMIN_USER), strtolower(trim((string) g($req, 'au', '')))) && check_pw(array('salt' => ADMIN_SALT, 'hash' => ADMIN_HASH, 'iter' => 60000), g($req, 'apw', ''));
  if (!$ok) { @file_put_contents(data_dir() . '/admin.log', date('c') . ' failed admin login from ' . client_ip() . "\n", FILE_APPEND); fail(401, 'bad_admin', 'That admin name or password is wrong.'); }
  $u = clean_user(g($req, 'u')); $acct = load_account($u);
  if (!$acct) fail(404, 'no_user', 'There is no account with that username.');
  $acct['pw'] = make_pw(clean_pw(g($req, 'pw'))); $acct['fails'] = 0; $acct['lock_until'] = 0; save_account($u, $acct);
  with_lock(acct_dir($u) . '/.tokens.lock', function () use ($u) { write_json(tokens_path($u), array(), true); });
  @file_put_contents(data_dir() . '/admin.log', date('c') . " temporary password set for $u from " . client_ip() . "\n", FILE_APPEND);
  return array('ok' => true);
}

=== api.php: act_reset ===
function act_reset($req) {
  $u = clean_user(g($req, 'u')); $pw = clean_pw(g($req, 'pw'));
  rate_limit('reset', 10, 3600);
  $acct = load_account($u);
  $ok = $acct && !empty($acct['reset_key']) && (int) $acct['reset_exp'] > now_ms() && hash_equals($acct['reset_key'], sha256(g($req, 'key', '')));
  if (!$ok) fail(400, 'bad_reset', 'That reset link is not valid or has expired. Ask for a new one.');
  $acct['pw'] = make_pw($pw); unset($acct['reset_key'], $acct['reset_exp']);
  $acct['confirmed'] = true; $acct['fails'] = 0; $acct['lock_until'] = 0;
  save_account($u, $acct);
  with_lock(acct_dir($u) . '/.tokens.lock', function () use ($u) { write_json(tokens_path($u), array(), true); });
  return array('ok' => true);
}

  acct['lock_until'] = 0
  save_account(u, acct)
  with_lock(File.join(acct_dir(u), '.tokens.lock')) { write_json(tokens_path(u), {}) }
  File.open(File.join(DATA_DIR, 'admin.log'), 'a') { |f| f.write("#{Time.now} temporary password set for #{u} from #{client_ip}\n") } rescue nil
  { 'ok' => true }

def act_forgot(req)
  u = clean_user(req['u'])
  email = req['email'].to_s.strip.downcase
  rate_limit!('forgot', 5, 3600)
  acct = load_account(u)
  if acct && !email.empty? && !acct['email'].to_s.empty? && acct['email'].to_s.downcase == email
    key = SecureRandom.hex(24)
    acct['reset_key'] = sha256(key)
    acct['reset_exp'] = now_ms + RESET_TTL * 1000
    save_account(u, acct)
    base = APP_URL.sub(%r{/*\z}, '/')
    send_mail(acct['email'], 'Reset your Aiu Totem password',
              "Hi #{u},\n\nOpen this link within 1 hour to choose a new password:\n\n#{base}#reset/#{u}/#{key}\n\n" \
              "If you did not ask for this, you can ignore it. Your password has not changed.\n")
  end
  { 'ok' => true } # same answer either way, so nobody can fish for accounts
end

=== api.cgi: act_reset ===
def act_reset(req)
  u = clean_user(req['u'])
  pw = clean_pw(req['pw'])
  rate_limit!('reset', 10, 3600)
  acct = load_account(u)
  ok = acct && acct['reset_key'] && acct['reset_exp'].to_i > now_ms && secure_eq(sha256(req['key']), acct['reset_key'])
  fail!(400, 'bad_reset', 'That reset link is not valid or has expired. Ask for a new one.') unless ok
  acct['pw'] = make_pw(pw)
  acct.delete('reset_key')
  acct.delete('reset_exp')
  acct['confirmed'] = true # they proved they own the email
  acct['fails'] = 0
  acct['lock_until'] = 0
  save_account(u, acct)
  with_lock(File.join(acct_dir(u), '.tokens.lock')) { write_json(tokens_path(u), {}) } # every device must log in again
  { 'ok' => true }
end

=== js/app.js: rescueMake / rescueUse ===
  /* 2.6.0: the recovery code (see vault.js). Made on this device when you ask (Settings > User info, or the note in
     Accounts > Cards), shown once to write down. With it, a temporary password from the admin opens everything again. */
  function rescueRec() { var r = app.S && C.setting(app.S, 'vaultRescue', null); return r && r.ct ? r : null; }
  app.hasRescue = function () { return !!rescueRec(); };
  app.rescueMake = function () {
    var V = AIU.vault;
    return app.vaultKey().then(function (k) {
      if (!k) throw new Error('Open Accounts with your Aiu password first.');
      var code = V.newCode();
      return V.wrapRescue(k, code).then(function (rec) { return app.addEvents([{ type: 'set', data: { k: 'vaultRescue', v: rec } }]); }).then(function () { return code; });
    });
  };
  app.rescueUse = function (code, pw) { // your recovery code + the password you log in with now
    var V = AIU.vault, rec = rescueRec();
    return (app.prof.token ? app.syncNow('rescue').catch(function () {}) : Promise.resolve()).then(function () {
      rec = rescueRec(); var lock = lockNow();
      if (!rec || !lock) throw new Error('This Aiu has no recovery code yet.');
      return V.openRescue(code, rec).then(function (k) {
        return V.fits(k, lock).then(function (ok) { if (!ok) throw new Error('That code is from before a password change. Use the newest code you wrote down.'); return keep(k, lock); });
      }).then(function () { return app.vaultRekey(pw); });
    });
  };

=== js/vault.js ===
  /* 2.6.0: the recovery code. A code you write down (16 letters and numbers) wraps your card key, so if your password is
     ever lost (the admin sets a temporary one), the code opens your cards and files again. The code itself is kept
     only scrambled with your card key, so a new password can re-wrap the same code; the server never sees it. */
  var CODE_CH = 'ABCDEFGHJKMNPQRSTUVWXYZ23456789';
  V.newCode = function () { var u = rand(16), out = ''; for (var i = 0; i < 16; i++) { out += CODE_CH.charAt(u[i] % CODE_CH.length); if (i % 4 === 3 && i < 15) out += '-'; } return out; };
  V.cleanCode = function (c) { var t = String(c || '').toUpperCase().replace(/[^A-Z0-9]/g, ''); return t.replace(/(.{4})(?=.)/g, '$1-'); };
  V.wrapRescue = function (key, code) { // -> {salt, iv, ct (the key, wrapped by the code), civ, cct (the code, sealed by the key)}
    var salt = rand(16);
    return Promise.all([V.exportKey(key), derive('rescue:' + V.cleanCode(code), salt)]).then(function (r) {
      return Promise.all([seal(r[1], r[0]), seal(key, code)]).then(function (x) { return { salt: b64(salt), iv: x[0].iv, ct: x[0].ct, civ: x[1].iv, cct: x[1].ct, rounds: ROUNDS }; });
    });
  };
  V.openRescue = function (code, rec) { // -> the card key, or rejects
    return derive('rescue:' + V.cleanCode(code), unb64(rec.salt), rec.rounds).then(function (k) { return open(k, rec.iv, rec.ct); }).then(function (raw) { return V.importKey(raw); }, function () { throw new Error('That recovery code doesn\'t fit. Check it and try again.'); });
  };
  V.codeOf = function (key, rec) { return open(key, rec.civ, rec.cct); }; // the code, read back with the card key (to re-wrap it after a new password)
